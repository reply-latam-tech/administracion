/**
 * Genera un token OAuth de Google internamente para consultar Firestore SIN LIBRERÃAS EXTRAS.
 */
function getFirestoreToken() {
  const header = Utilities.base64EncodeWebSafe(JSON.stringify({alg: "RS256", typ: "JWT"}));
  const now = Math.floor(Date.now() / 1000);
  const claimSet = {
    iss: FIRESTORE_CONFIG.email,
    scope: "https://www.googleapis.com/auth/datastore",
    aud: "https://oauth2.googleapis.com/token",
    exp: now + 3600,
    iat: now
  };
  const payload = Utilities.base64EncodeWebSafe(JSON.stringify(claimSet));
  const signature = Utilities.computeRsaSha256Signature(header + "." + payload, FIRESTORE_CONFIG.key);
  const jwt = header + "." + payload + "." + Utilities.base64EncodeWebSafe(signature);
  
  const response = UrlFetchApp.fetch("https://oauth2.googleapis.com/token", {
    method: "post",
    payload: { grant_type: "urn:ietf:params:oauth:grant-type:jwt-bearer", assertion: jwt }
  });
  return JSON.parse(response.getContentText()).access_token;
}

/**
 * FunciÃ³n que itera sobre los sellers en Firestore y renueva sus tokens.
 */
function checkAndRefreshTokens() {
  Logger.log("Iniciando revisiÃ³n y refresco de tokens...");
  
  const firestoreToken = getFirestoreToken();
  const url = `https://firestore.googleapis.com/v1/projects/${FIRESTORE_CONFIG.projectId}/databases/(default)/documents/accounts?pageSize=1000`;
  
  const response = UrlFetchApp.fetch(url, {
    method: "get",
    headers: { "Authorization": `Bearer ${firestoreToken}` }
  });
  
  const data = JSON.parse(response.getContentText());
  if (!data.documents) return;
  
  data.documents.forEach(doc => {
    const fields = doc.fields || {};
    const sellerId = fields.seller_id_meli ? fields.seller_id_meli.integerValue || fields.seller_id_meli.stringValue : null;
    const refreshToken = fields.refresh_token ? fields.refresh_token.stringValue : null;
    const active = fields.active ? fields.active.booleanValue : false;
    
    if (refreshToken && active) {
      Logger.log(`Refrescando token para el seller ${sellerId}...`);
      try {
        const newTokens = refreshMeliToken(refreshToken);
        
        // Actualizar en Firestore
        const patchUrl = `https://firestore.googleapis.com/v1/${doc.name}?updateMask.fieldPaths=access_token&updateMask.fieldPaths=refresh_token&updateMask.fieldPaths=last_update`;
        const patchPayload = {
          fields: {
            access_token: { stringValue: newTokens.access_token },
            refresh_token: { stringValue: newTokens.refresh_token },
            last_update: { stringValue: new Date().toISOString() }
          }
        };
        
        UrlFetchApp.fetch(patchUrl, {
          method: "patch",
          headers: { "Authorization": `Bearer ${firestoreToken}`, "Content-Type": "application/json" },
          payload: JSON.stringify(patchPayload)
        });
        
        Logger.log(`Tokens renovados exitosamente para ${sellerId}`);
      } catch (e) {
        Logger.log(`Error al renovar token de ${sellerId}: ${e.message}`);
      }
    }
  });
}

/**
 * Llama a la API de Mercado Libre para obtener nuevos tokens.
 */
function refreshMeliToken(refreshToken) {
  const url = "https://api.mercadolibre.com/oauth/token";
  
  const payload = {
    grant_type: "refresh_token",
    client_id: CONFIG.MELI_CLIENT_ID,
    client_secret: CONFIG.MELI_CLIENT_SECRET,
    refresh_token: refreshToken
  };
  
  const options = {
    method: "post",
    contentType: "application/x-www-form-urlencoded",
    payload: payload,
    muteHttpExceptions: true
  };
  
  const response = UrlFetchApp.fetch(url, options);
  const json = JSON.parse(response.getContentText());
  
  if (response.getResponseCode() !== 200) {
    throw new Error(json.message || "Error desconocido al refrescar token");
  }
  
  return json;
}

/**
 * Función utilitaria para obtener los sellers activos desde Firestore (nativo).
 * Vuelve a cruzar con la hoja "seller" para extraer ÚNICAMENTE las cuentas
 * listadas allí y con el nombre personalizado (nickname).
 */
function getActiveSellers() {
  const allowedSellers = {};
  let hasAllowedSellers = false;
  
  try {
    const ss = SpreadsheetApp.openById(CONFIG.SPREADSHEET_ID);
    const sheetSeller = ss.getSheetByName(CONFIG.SHEET_SELLER || "seller");
    
    if (sheetSeller) {
      const lastRow = sheetSeller.getLastRow();
      if (lastRow > 1) {
        const data = sheetSeller.getRange(2, 1, lastRow - 1, 2).getValues();
        data.forEach(row => {
          const id = String(row[0]).trim();
          const sellerName = String(row[1]).trim();
          if (id && sellerName) {
            allowedSellers[id] = sellerName;
            hasAllowedSellers = true;
          }
        });
      }
    }
  } catch (e) {
    Logger.log("[AUTH] Error leyendo hoja seller: " + e.message);
  }

  const firestoreToken = getFirestoreToken();
  const url = `https://firestore.googleapis.com/v1/projects/${FIRESTORE_CONFIG.projectId}/databases/(default)/documents/accounts?pageSize=1000`;
  
  const response = UrlFetchApp.fetch(url, {
    method: "get",
    headers: { "Authorization": `Bearer ${firestoreToken}` }
  });
  
  const data = JSON.parse(response.getContentText());
  const sellers = [];
  
  if (data.documents) {
    data.documents.forEach(doc => {
      const fields = doc.fields || {};
      const active = fields.active ? fields.active.booleanValue : false;
      const accessToken = fields.access_token ? fields.access_token.stringValue : null;
      
      if (active && accessToken) {
        const rawId = fields.seller_id_meli ? fields.seller_id_meli.integerValue || fields.seller_id_meli.stringValue : null;
        const idString = String(rawId).trim();
        
        // Si encontramos la lista de sellers permitidos, solo agregamos los que están ahí
        if (rawId && hasAllowedSellers) {
          if (allowedSellers[idString]) {
            sellers.push({
              seller_id_meli: idString,
              nickname: allowedSellers[idString], // Usamos el nombre de tu hoja
              access_token: accessToken
            });
          }
        } else if (rawId && !hasAllowedSellers) {
          // Fallback en caso de que no hayas creado la hoja 'seller' todavía
          const nicknameFb = fields.nickname ? fields.nickname.stringValue : `Seller_${idString}`;
          sellers.push({
            seller_id_meli: idString,
            nickname: nicknameFb, 
            access_token: accessToken
          });
        }
      }
    });
  }
  
  return sellers;
}
